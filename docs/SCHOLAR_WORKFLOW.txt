name: Update Scholar map and publish website
on:
  schedule:
    - cron: '23 8 * * *'
  workflow_dispatch:
  push:
    branches: [main]
permissions:
  contents: read
concurrency:
  group: scholar-pages
  cancel-in-progress: false
jobs:
  build:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v6
        with:
          ref: main
      - name: Read Scholar and update map
        if: github.event_name != 'push'
        env:
          SERPAPI_API_KEY: ${{ secrets.SERPAPI_API_KEY }}
        run: python3 scripts/sync_scholar.py
      - name: Save updated publication records
        if: github.event_name != 'push'
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add publications-map.js scripts/scholar-state.json
          if ! git diff --cached --quiet; then
            git commit -m 'Update Scholar publication map'
            git push origin HEAD:main
          fi
      - name: Stage public website files only
        run: |
          python3 - <<'PY'
          from pathlib import Path
          import shutil
          output = Path('_site')
          output.mkdir(exist_ok=True)
          extensions = {'.html','.css','.js','.svg','.png','.jpg','.jpeg','.webp','.gif','.ico','.pdf','.woff','.woff2','.xml','.txt'}
          for file in Path('.').iterdir():
              if file.is_file() and not file.is_symlink() and (file.suffix.lower() in extensions or file.name in {'.nojekyll','CNAME'}):
                  shutil.copy2(file, output / file.name)
          assert (output / 'index.html').exists()
          PY
      - uses: actions/upload-pages-artifact@v4
        with:
          path: _site
  deploy:
    needs: build
    runs-on: ubuntu-latest
    permissions:
      pages: write
      id-token: write
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - name: Publish to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4
